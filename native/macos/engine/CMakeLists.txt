cmake_minimum_required(VERSION 3.20)
project(GodEyeScene LANGUAGES C CXX OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CESIUM_TESTS_ENABLED OFF CACHE BOOL "" FORCE)
set(CESIUM_DISABLE_CURL ON CACHE BOOL "" FORCE)
set(CESIUM_DISABLE_LIBJPEG_TURBO ON CACHE BOOL "" FORCE)
set(CESIUM_ENABLE_CLANG_TIDY OFF CACHE BOOL "" FORCE)
add_subdirectory("${GODEYE_CESIUM_SOURCE}" cesium)
# Bound PNG/JPEG decoding before allocating texture pixels.
target_compile_definitions(CesiumImage PRIVATE STBI_MAX_DIMENSIONS=8192)
add_library(GodEyeScene STATIC GodEyeScene.mm ../../scene/Scene.cpp)
target_include_directories(GodEyeScene PRIVATE ../Sources/CScene/include ../../scene)
target_compile_options(GodEyeScene PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
target_link_libraries(GodEyeScene PRIVATE Cesium3DTilesSelection CesiumGltfContent Cesium3DTilesContent "-framework Foundation")
