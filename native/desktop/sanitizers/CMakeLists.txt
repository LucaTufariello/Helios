cmake_minimum_required(VERSION 3.24)
project(HeliosSanitizerChecks LANGUAGES C CXX)
enable_testing()
set(HELIOS_BUILD_DESKTOP ON CACHE BOOL "" FORCE)
set(BUILD_TESTING ON CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../.." "${CMAKE_BINARY_DIR}/project" EXCLUDE_FROM_ALL)

# Instrument owned storage/network/process/geometry code and its tests without
# changing the production build or recompiling all static upstream scene code.
function(helios_instrument directory)
  get_property(helios_targets DIRECTORY "${directory}" PROPERTY BUILDSYSTEM_TARGETS)
  foreach(helios_target IN LISTS helios_targets)
    if(helios_target MATCHES "^Helios")
      get_target_property(helios_type "${helios_target}" TYPE)
      if(helios_type STREQUAL "STATIC_LIBRARY" OR helios_type STREQUAL "EXECUTABLE")
        target_compile_options("${helios_target}" PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        if(helios_type STREQUAL "EXECUTABLE")
          target_link_options("${helios_target}" PRIVATE -fsanitize=address,undefined)
        endif()
      endif()
      if(helios_target MATCHES "Tests$" AND NOT helios_target STREQUAL "HeliosSceneTests")
        file(APPEND "${CMAKE_BINARY_DIR}/sanitizer-targets.txt" "${helios_target}\n")
      endif()
    endif()
  endforeach()
  get_property(helios_children DIRECTORY "${directory}" PROPERTY SUBDIRECTORIES)
  foreach(helios_child IN LISTS helios_children)
    helios_instrument("${helios_child}")
  endforeach()
endfunction()
file(WRITE "${CMAKE_BINARY_DIR}/sanitizer-targets.txt" "")
helios_instrument("${CMAKE_CURRENT_SOURCE_DIR}/../../..")
