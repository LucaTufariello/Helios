cmake_minimum_required(VERSION 3.24)
project(GodEyeSanitizerChecks LANGUAGES C CXX)
enable_testing()
set(GODEYE_BUILD_DESKTOP ON CACHE BOOL "" FORCE)
set(BUILD_TESTING ON CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../.." "${CMAKE_BINARY_DIR}/project" EXCLUDE_FROM_ALL)

# Instrument owned storage/network/process/geometry code and its tests without
# changing the production build or recompiling all static upstream scene code.
function(godeye_instrument directory)
  get_property(godeye_targets DIRECTORY "${directory}" PROPERTY BUILDSYSTEM_TARGETS)
  foreach(godeye_target IN LISTS godeye_targets)
    if(godeye_target MATCHES "^GodEye")
      get_target_property(godeye_type "${godeye_target}" TYPE)
      if(godeye_type STREQUAL "STATIC_LIBRARY" OR godeye_type STREQUAL "EXECUTABLE")
        target_compile_options("${godeye_target}" PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        if(godeye_type STREQUAL "EXECUTABLE")
          target_link_options("${godeye_target}" PRIVATE -fsanitize=address,undefined)
        endif()
      endif()
      if(godeye_target MATCHES "Tests$" AND NOT godeye_target STREQUAL "GodEyeSceneTests")
        file(APPEND "${CMAKE_BINARY_DIR}/sanitizer-targets.txt" "${godeye_target}\n")
      endif()
    endif()
  endforeach()
  get_property(godeye_children DIRECTORY "${directory}" PROPERTY SUBDIRECTORIES)
  foreach(godeye_child IN LISTS godeye_children)
    godeye_instrument("${godeye_child}")
  endforeach()
endfunction()
file(WRITE "${CMAKE_BINARY_DIR}/sanitizer-targets.txt" "")
godeye_instrument("${CMAKE_CURRENT_SOURCE_DIR}/../../..")
